  return h;
}

function bruteForce(prefix) {
  for (let len = 0; len <= 8; len++) {
    for (let attempt = 0; attempt < 3000000; attempt++) {
      let salt = '';
      for (let i = 0; i < len; i++) salt += chars[Math.floor(Math.random()*chars.length)];
      const candidate = base + salt;
      const h = hash(candidate).toString(16).padStart(8,'0');
      if (h.startsWith(prefix)) return { candidate, sig: h };
    }
  }
  return null;
}
